LAB 01

Módulo Interativo de ETL & Tratamento de Dados

Pipeline de Sanitização & Validação de Dados: tratamento automatizado de CPFs/CNPJs, nulos e duplicados, padronização de datas, regex para e-mails e telefones e geração de métricas de qualidade (Data Quality Check).

Formato: nome,cpf,email,telefone,nascimento,valor — cole os seus dados sujos ou edite a amostra.
Antes bruto, como veio da origem
Depois padronizado e validado
python · pandas + polars
# ============================================================
# ETL · Sanitização & Validação de Dados
# leitura crua -> padronização -> validação -> Data Quality
# ============================================================
import re
import pandas as pd
import polars as pl

EMAIL_RE = re.compile(r"^[A-Za-z._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$")

def only_digits(v):
    """Remove tudo que não for dígito (máscara de CPF/CNPJ)."""
    return re.sub(r"\D", "", str(v))

def is_valid_cpf(v):
    """Valida os dígitos verificadores (mod 11) do CPF."""
    c = only_digits(v)
    if len(c) != 11 or c == c[0] * 11:
        return False
    dv = lambda s, w: sum(int(a) * b for a, b in zip(s, w)) % 11
    d1 = dv(c[:9], range(10, 1, -1));  d1 = 0 if d1 < 2 else 11 - d1
    d2 = dv(c[:10], range(11, 1, -1)); d2 = 0 if d2 < 2 else 11 - d2
    return c[9] == str(d1) and c[10] == str(d2)

def mask_cpf(v):
    """12345678900 -> 123.445.678-90 (None se o tamanho não bate)."""
    c = only_digits(v)
    return f"{c[:3]}.{c[3:6]}.{c[6:9]}-{c[9:]}" if len(c) == 11 else None

def norm_phone(v):
    """(12) 98169-4128 | 12981694128 -> máscara padrão."""
    d = only_digits(v)
    return f"({d[:2]}) {d[2:-4]}-{d[-4:]}" if len(d) in (10, 11) else None

def norm_date(v):
    """dd/mm/yyyy, dd-mm-yyyy ou yyyy-mm-dd -> ISO (yyyy-mm-dd)."""
    s = str(v).strip()
    for fmt in ("%d/%m/%Y", "%d-%m-%Y", "%Y-%m-%d"):
        try:
            return pd.to_datetime(s, format=fmt).date().isoformat()
        except (ValueError, TypeError):
            continue
    return None

def money(v):
    """'R$ 1.234,56' | '899,90' | '1500' -> float (padrão pt-BR)."""
    s = re.sub(r"[^\d,.-]", "", str(v))
    if not s:
        return None
    if "," in s:
        s = s.replace(".", "").replace(",", ".")
    try:
        return float(s)
    except ValueError:
        return None

def clean_pipeline(df):
    """Saneamento completo + relatório de qualidade (DQ report)."""
    report = {}
    df = df.apply(lambda c: c.astype(str).str.strip())      # 1. trim
    df = df.replace({"": None, "nan": None, "None": None})  # 2. nulos reais
    report["nulos"] = int(df.isna().sum().sum())

    n = len(df)
    df = df.drop_duplicates()                               # 3. duplicados
    report["duplicados"] = n - len(df)

    df["nome"] = df["nome"].str.title()                     # 4. padroniza
    df["cpf"] = df["cpf"].map(mask_cpf)                     # 5. máscara
    df["cpf_invalido"] = ~df["cpf"].map(lambda v: bool(v) and is_valid_cpf(v))
    df.loc[df["cpf_invalido"], "cpf"] = None                # registro reprovado
    df["email"] = df["email"].str.lower()
    df["email_invalido"] = ~df["email"].fillna("").str.match(EMAIL_RE)
    df["telefone"] = df["telefone"].map(norm_phone)         # 6. regex
    df["nascimento"] = df["nascimento"].map(norm_date)      # 7. datas
    df["valor"] = df["valor"].map(money)

    report["cpf_invalido"] = int(df["cpf_invalido"].sum())
    report["email_invalido"] = int(df["email_invalido"].sum())
    report["datas_ok"] = int(df["nascimento"].notna().sum())
    return df, report

# ---------- execução ----------
raw = pd.read_csv("dados_sujos.csv", dtype=str)
clean, dq = clean_pipeline(raw)

pl.from_pandas(clean).write_parquet("dados_limpos.parquet")  # saída analítica
print("Data Quality Check:", dq)

Regra de ouroNenhum registro entra na camada analítica sem passar pelos checks. O pipeline mede, bloqueia e reporta — qualidade de dado é requisito, não detalhe de implementação.

Checks aplicados a cada carga
  • Completude — contagem de nulos por coluna; campos-chave (cpf, email) não podem ser nulos no load.
  • Unicidade — drop_duplicates() sobre a linha normalizada; duplicidade é medida antes de contar métricas de negócio.
  • Forma (regex) — e-mail contra EMAIL_RE e telefone contra a máscara (DD) 99999-9999.
  • Validade de domínio — dígitos verificadores do CPF (mod 11) e datas em intervalo plausível, sem futuro.
  • Tipagem — valores monetários pt-BR (1.234,56) convertidos para float antes do agregado.

def data_quality_gate(df):
    """Levanta exceção e interrompe o load se algum check falhar."""
    checks = {
        "nulos_em_chave":  df["cpf"].isna().sum() == 0,
        "unicidade":       not df.duplicated().any(),
        "regex_email":     df["email"].fillna("").str.match(EMAIL_RE).all(),
        "intervalo_valor": df["valor"].between(0, 1_000_000).all(),
        "sem_data_futura": pd.to_datetime(df["nascimento"]).max()
                           < pd.Timestamp.today(),
    }
    failed = [k for k, ok in checks.items() if not ok]
    if failed:
        raise ValueError(f"Data Quality Check falhou: {failed}")
    return True

Laboratório Didático de SQL Avançado & Analytics Engineering

SQL Analytics Lab: consultas avançadas cobrindo Window Functions (ROW_NUMBER, LAG/LEAD, NTILE), CTEs encadeadas, agregações complexas e otimização de queries para ecossistemas analíticos.

Cenário de negócio Operação de e-commerce com 24 meses de histórico em modelo dimensional (fato_vendas, dim_cliente, dim_tempo). O time comercial precisa entender retenção por coorte, calcular LTV e churn e detectar queda de receita no dia seguinte. Cada aba mostra a consulta, o porquê de cada função e o resultado da simulação.

sql · postgres / t-sql
-- Média móvel de 7 dias + acumulado, em uma única passada
WITH vendas_dia AS (
    SELECT  t.data_pedido,
            SUM(v.valor_total) AS receita_dia
    FROM    fato_vendas v
    JOIN    dim_tempo t ON t.sk_tempo = v.sk_tempo
    GROUP BY t.data_pedido
)
SELECT  data_pedido,
        receita_dia,
        AVG(receita_dia) OVER (
            ORDER BY data_pedido
            ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
        ) AS media_movel_7d,
        SUM(receita_dia) OVER (
            ORDER BY data_pedido
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) AS acumulado
FROM    vendas_dia
ORDER BY data_pedido;
Por que assim — explicação linha a linha
  • CTE primeiro: vendas_dia fixa a granularidade diária antes da janela — a agregação roda uma vez, e não a cada linha do resultado.
  • AVG() OVER (ORDER BY … ROWS BETWEEN 6 PRECEDING AND CURRENT ROW): janela deslizante fixa em linhas = média dos últimos 7 registros sem self-join correlacionada.
  • ROWS e não RANGE: RANGE agruparia datas empatadas e inflaria a janela; ROWS conta posições físicas e custa menos memória.
  • Acumulado na mesma passada: SUM() OVER (… UNBOUNDED PRECEDING) resolve o running total sem segunda CTE nem variável de sessão.
  • Por que não subquery: a versão correlacionada reprocessa a CTE para cada linha (O(n²)); a janela ordena uma vez e varre em O(n log n).

sql · top 3 por região + quartil de LTV
-- Ranking regional e bucket de LTV em uma única consulta
WITH receita_cliente AS (
    SELECT  c.sk_cliente,
            c.regiao,
            c.nome_cliente,
            SUM(v.valor_total)  AS receita_90d,
            COUNT(DISTINCT v.sk_pedido) AS pedidos
    FROM    fato_vendas v
    JOIN    dim_cliente c ON c.sk_cliente = v.sk_cliente
    WHERE   v.data_pedido >= CURRENT_DATE - INTERVAL '90 days'
    GROUP BY c.sk_cliente, c.regiao, c.nome_cliente
),
ranked AS (
    SELECT  *,
            ROW_NUMBER() OVER (
                PARTITION BY regiao ORDER BY receita_90d DESC
            ) AS rn_regiao,
            NTILE(4) OVER (ORDER BY receita_90d DESC) AS quartil_ltv
    FROM    receita_cliente
)
SELECT  regiao, nome_cliente, receita_90d, pedidos, quartil_ltv
FROM    ranked
QUALIFY rn_regiao <= 3
ORDER BY regiao, receita_90d DESC;
Por que assim — explicação linha a linha
  • CTE em dois estágios: primeiro agrega para a métrica de negócio (receita_90d), depois ranqueia — separar agregação de classificação torna cada passo testável.
  • PARTITION BY regiao: zera o ranking a cada região; sem ele o ROW_NUMBER seria global e esconderia os top de cada praça.
  • NTILE(4): divide a base em quartis de LTV já ordenados — pronto para campanha de retenção (Q1 = churn iminente, Q4 = melhor conta).
  • QUALIFY em vez de subquery: filtra o resultado da janela sem abrir mais um nível de leitura; o plano consegue avaliar o filtro junto com a partição.
  • Portabilidade: QUALIFY é nativo de Snowflake/BigQuery/Redshift. Em PostgreSQL/SQL Server a equivalência é envolver a CTE ranked e aplicar WHERE rn_regiao <= 3 — mesmo plano, sem janela duplicada.

-- Variação D-1 e desvio da média de 30 dias (janela que exclui o dia corrente)
WITH diario AS (
    SELECT  t.data_pedido,
            SUM(v.valor_total) AS receita
    FROM    fato_vendas v
    JOIN    dim_tempo t ON t.sk_tempo = v.sk_tempo
    GROUP BY t.data_pedido
),
com_variacao AS (
    SELECT  *,
            LAG(receita, 1) OVER (ORDER BY data_pedido) AS receita_d1,
            receita - LAG(receita, 1) OVER (ORDER BY data_pedido) AS delta,
            AVG(receita) OVER (
                ORDER BY data_pedido
                ROWS BETWEEN 30 PRECEDING AND 1 PRECEDING
            ) AS media_30d
    FROM    diario
)
SELECT  data_pedido,
        receita,
        receita_d1,
        ROUND(100.0 * delta / NULLIF(receita_d1, 0), 1) AS variacao_pct,
        ROUND(media_30d, 2) AS media_30d,
        CASE
            WHEN receita < media_30d * 0.6 THEN 'QUEDA ANOMALA'
            WHEN receita > media_30d * 1.6 THEN 'PICO ATIPICO'
            ELSE 'Normal'
        END AS status
FROM    com_variacao
WHERE   media_30d IS NOT NULL
ORDER BY data_pedido;
Por que assim — explicação linha a linha
  • LAG(receita, 1): traz o valor da linha anterior dentro do mesmo resultado — substitui o self-join clássico t1.data = t2.data - 1 com custo de ordenação única.
  • NULLIF(receita_d1, 0): anula a divisão quando não há D-1; sem isso, o primeiro dia do histórico quebra a query com division by zero.
  • ROWS BETWEEN 30 PRECEDING AND 1 PRECEDING: a média de referência exclui o dia corrente — incluí-lo criaria vazamento (leakage) e suavizaria justamente o pico que queremos flagrar.
  • CASE com limiares relativos: 60% abaixo / 60% acima da média move junto com a sazonalidade da base, em vez de um número fixo de reais.
  • Otimização: uma única ordenação atende as três janelas (LAG, aritmética e AVG) — o planner reutiliza o mesmo Sort + WindowAgg.

with pedidos as (
    select * from {{ ref('stg_pedidos') }}
    {% if is_incremental() %}
      where atualizado_em > (select coalesce(max(atualizado_em), '1900-01-01')
                              from {{ this }})
    {% endif %}
),
itens as (
    select sk_pedido, sk_produto, sk_item, qtd, preco_unitario
    from {{ ref('stg_itens_pedido') }}
)
select
    p.sk_pedido || '-' || i.sk_item::text as sk_venda,
    c.sk_cliente,
    pr.sk_produto,
    t.sk_tempo,
    i.qtd                                  as qtd_itens,
    i.qtd * i.preco_unitario               as receita,
    case when p.canal = 'marketplace'
         then 'Marketplace' else 'Direto' end as canal,
    p.atualizado_em
from pedidos p
join itens i                       on i.sk_pedido   = p.sk_pedido
join {{ ref('dim_cliente') }} c    on c.nk_cliente  = p.id_cliente
                                  and c.is_current          -- 1 linha vigente
join {{ ref('dim_produto') }} pr   on pr.nk_produto = i.id_produto
join {{ ref('dim_tempo') }}   t    on t.data_pedido = p.data_pedido
sql · dbt · dim_cliente.sql (SCD Type 2)
{{ config(materialized = 'table') }}

-- DIMENSÃO SCD TYPE 2 · preserva o histórico de mudanças
with base as (
    select *,
           md5(concat_ws('|', nome, regiao, segmento)) as hash_attr
    from {{ ref('stg_clientes') }}
),
vigente as (select * from {{ this }} where is_current),
delta as (
    select b.* from base b
    join vigente v on v.nk_cliente = b.nk_cliente
    where b.hash_attr <> v.hash_attr          -- só o que mudou de fato
),
novos as (
    select b.* from base b
    left join vigente v on v.nk_cliente = b.nk_cliente
    where v.nk_cliente is null
),
fechados as (                                   -- encerra a linha anterior
    select v.sk_cliente, v.nk_cliente, v.nome, v.regiao, v.segmento,
           v.valid_from, current_timestamp as valid_to, false as is_current
    from vigente v join delta d on d.nk_cliente = v.nk_cliente
),
abertos as (                                    -- abre a nova vigência
    select {{ dbt_utils.generate_surrogate_key(['nk_cliente','hash_attr']) }}
               as sk_cliente,
           b.nk_cliente, b.nome, b.regiao, b.segmento,
           current_timestamp as valid_from, null::timestamp as valid_to,
           true as is_current
    from (select * from delta union all select * from novos) b
)
select * from fechados union all select * from abertos
Por que assim
  • Surrogate key (sk_*): chave interna estável — a chave de negócio muda (reimportação, correção), a SK não, então histórico de vendas nunca fica órfão.
  • Hash de atributos: md5(concat_ws(...)) compara a linha inteira de uma vez; comparação barata e sem lista de colunas quebrada quando o schema muda.
  • Is incremental: só processa atualizado_em > max(...) — custo proporcional à novidade, não ao tamanho da tabela.
  • and c.is_current no join: garante 1 linha por cliente na resposta corrente; para análise histórica, troca-se por predicado de janela (valid_from <= data < valid_to).
yaml · dbt · schema.yml
# dbt/schema.yml — integridade referencial e regras de negócio
version: 2

models:
  - name: fct_vendas
    description: "Fato de vendas — granularidade: item de pedido."
    columns:
      - name: sk_venda
        tests: [unique, not_null]
      - name: sk_cliente
        tests:
          - relationships:
              to: ref('dim_cliente')
              field: sk_cliente        # FK -> PK (integridade referencial)
      - name: sk_produto
        tests:
          - relationships:
              to: ref('dim_produto')
              field: sk_produto
      - name: receita
        tests:
          - dbt_utils.expression_is_true:
              expression: ">= 0"     # receita nunca negativa

  - name: dim_cliente
    description: "SCD Type 2 — exatamente 1 linha vigente por cliente."
    tests:
      - dbt_utils.expression_is_true:
          expression: >-
            count(*) filter (where is_current)
            <= count(distinct nk_cliente)
    columns:
      - name: sk_cliente
        tests: [unique, not_null]
      - name: valid_to
        tests:
          - dbt_utils.expression_is_true:
              expression: "is null or valid_to > valid_from"

Como ler Retentativa = a tarefa reentrou na fila de retry e ainda pode fechar dentro do SLA. Falha = estourou as 3 tentativas e disparou o webhook de alerta. A barra vermelha é a janela que consumiu a folga do contrato de dados.

python · airflow dag
# ============================================================
# DAG de ingestão de vendas · orquestrador + alertas de SLA
# ============================================================
from datetime import datetime, timedelta
from airflow import DAG
from airflow.operators.python import PythonOperator
from airflow.providers.slack.hooks.slack_webhook import SlackWebhookOperator

DEFAULT_ARGS = {
    "owner": "data-eng",
    "retries": 3,
    "retry_delay": timedelta(minutes=10),
    "email_on_failure": True,
    "email": ["data@empresa.com"],
    "sla": timedelta(minutes=45),        # contrato de dados D-1 07:00
}

def on_failure_alert(context):
    """Webhook para o canal #data-alerts com o contexto da falha."""
    SlackWebhookOperator(
        task_id="alerta_falha",
        slack_webhook_conn_id="slack_webhook",
        text=(
            f":red_circle: *DAG `{context['dag'].dag_id}` falhou*\n"
            f"execucao: `{context['execution_date']}`\n"
            f"tarefa: `{context['task_instance'].task_id}`\n"
            f"log: {context['task_instance'].log_url}"
        ),
    ).execute(context=context)

def extract(**context):
    """Baixa a particao D-1 da API e grava na camada Bronze."""
    ...

def transform(**context):
    """Sanitiza, deduplica e grava a particao Silver (Delta)."""
    ...

def load_and_check(**context):
    """Recarrega o Gold, publica volumetria e valida a contagem."""
    ...

with DAG(
    dag_id="ingest_vendas_diaria",
    default_args=DEFAULT_ARGS,
    start_date=datetime(2026, 1, 1),
    schedule="0 6 * * *",               # 06:00, apos o fechamento D-1
    catchup=False,
    max_active_runs=1,                  # nao sobrepor particao em reprocesso
    on_failure_callback=on_failure_alert,
    tags=["sla-45min", "bronze-silver-gold"],
) as dag:
    t1 = PythonOperator(task_id="extract",       python_callable=extract)
    t2 = PythonOperator(task_id="transform",     python_callable=transform)
    t3 = PythonOperator(task_id="load_and_check", python_callable=load_and_check)
    t1 >> t2 >> t3
Por que assim
  • sla em DEFAULT_ARGS — o Airflow passa a marcar a execução como upstream SLA miss sozinho e agenda o alerta, sem cron externo.
  • on_failure_callback — o webhook recebe o log_url da tarefa; quem recebe a mensagem vai direto ao erro, sem navegar no scheduler.
  • retries=3 + retry_delay=10min — falha transitória de API não vira incidente; a coluna "retentativa" do painel é exatamente essa fila.
  • max_active_runs=1 — impede duas janelas escrevendo na mesma partição, a causa clássica de dado duplicado em lake.
  • catchup=False — deploy não dispara histórico acumulado de execuções em produção.

python · pyspark + delta lake
# ============================================================
# Lakehouse · Arquitetura Medallion (Bronze → Silver → Gold)
# ============================================================
from pyspark.sql import SparkSession, functions as F

spark = (SparkSession.builder
         .appName("lakehouse-medallion")
         .config("spark.sql.extensions",
                 "io.delta.sql.DeltaSparkSessionExtension")
         .config("spark.sql.catalog.spark_catalog",
                 "org.apache.spark.sql.delta.catalog.DeltaCatalog")
         .getOrCreate())

# ---------- BRONZE · ingestão crua, sem transformação ----------
bronze = (spark.read.format("json")
          .option("mode", "PERMISSIVE")     # registro quebrado vira null
          .load("s3://lake/bronze/eventos/"))

(bronze.write.format("parquet")
       .partitionBy("dt_chegada")           # particionamento = data
       .mode("append")
       .save("s3://lake/bronze/eventos/"))

# ---------- SILVER · limpeza, dedup e enriquecimento ----------
silver = (bronze
          .dropDuplicates(["id_evento", "versao"])        # deduplica
          .withColumn("evento", F.trim(F.lower("tipo_evento")))
          .withColumn("dt_evento", F.to_date("dt_evento"))
          .withColumn("valor", F.col("valor").cast("decimal(18,2)"))
          .filter(F.col("id_evento").isNotNull()))

(silver.write.format("delta")
       .partitionBy("dt_evento")
       .mode("overwrite")
       .option("replaceWhere", "dt_evento >= '2026-01-01'")   # overwrite seletivo
       .save("s3://lake/silver/eventos/"))

# ---------- GOLD · agregação de negócio para o Power BI ----------
gold = (silver.groupBy("dt_evento", "evento", "regiao")
        .agg(F.countDistinct("id_evento").alias("eventos"),
             F.count(F.lit(1)).alias("ocorrencias"),
             F.sum("valor").alias("valor_total")))

(gold.write.format("delta")
     .mode("overwrite")
     .save("s3://lake/gold/kpi_eventos"))
Por que assim
  • PERMISSIVE — arquivo malformado vira null em vez de derrubar o lote inteiro; o registro sujo é tratado na Silver, não perdido na Bronze.
  • partitionBy("dt_evento") — filtro por data lê só as partições pedidas (pruning), que é o que mantém o custo do Spark previsível.
  • Delta + replaceWhere — reprocessamento sobrescreve apenas o intervalo afetado, com transação atômica e histórico de versões (time travel).
  • dropDuplicates(["id_evento", "versao"]) — chave de idempotência: reexecutar a mesma janela não duplica métrica.
  • Gold só agrrega — nenhuma regra de limpeza mora na camada de consumo; o Power BI lê tabela pequena e estável.
